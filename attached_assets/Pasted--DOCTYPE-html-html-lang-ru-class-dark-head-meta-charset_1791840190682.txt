    <script>
        let isRain = true;
        function refreshWeather() {
            const icon = document.getElementById('weather-icon');
            const text = document.getElementById('weather-text');
            const advice = document.getElementById('weather-advice');
            if (isRain) {
                icon.className = "fa-solid fa-sun";
                text.innerText = "+22°C, Ясно";
                advice.innerText = "Совет: Отличная погода для прогулки в парке.";
                isRain = false;
            } else {
                icon.className = "fa-solid fa-cloud-rain";
                text.innerText = "+14°C, Дождь";
                advice.innerText = "Совет: После 15:00 ожидаются осадки, рекомендуется захватить зонт.";
                isRain = true;
            }
            showToast('Погода обновлена');
        }

        function triggerShazam() {
            const badge = document.getElementById('shazam-badge');
            const title = document.getElementById('track-title');
            const artist = document.getElementById('track-artist');
            const disc = document.getElementById('album-disc');

            badge.innerText = 'Слушаем...';
            disc.classList.add('animate-once-spin');
            setTimeout(() => {
                disc.classList.remove('animate-once-spin');
            }, 1000);

            setTimeout(() => {
                title.innerText = 'Midnight City (VARYAG Mix)';
                artist.innerText = 'M83 • Распознано Shazam';
                badge.innerText = 'Найдено';
                showToast('Трек добавлен в плейлист');
            }, 1500);
        }

        function connectStreaming(service) {
            const status = document.getElementById(`${service}-status`);
            const btn = document.getElementById(`${service}-btn`);
            status.innerText = 'Подключено (@user)';
            status.className = 'text-[10px] text-emerald-500 font-medium';
            btn.innerText = 'Отвязать';
            btn.onclick = () => disconnectStreaming(service);
            showToast('Аккаунт успешно подключен');
        }

        function disconnectStreaming(service) {
            const status = document.getElementById(`${service}-status`);
            const btn = document.getElementById(`${service}-btn`);
            status.innerText = 'Не подключено';
            status.className = 'text-[10px] opacity-50';
            btn.innerText = 'Подключить';
            btn.onclick = () => connectStreaming(service);
            showToast('Аккаунт отвязан');
        }

        function openLogModal(sender, time, text) {
            document.getElementById('modal-sender').innerText = sender;
            document.getElementById('modal-time').innerText = time;
            document.getElementById('modal-content').innerText = text;
            document.getElementById('log-modal').classList.remove('hidden');
        }

        function closeLogModal() {
            document.getElementById('log-modal').classList.add('hidden');
        }

        function toggleTheme() {
            const html = document.documentElement;
            const icon = document.getElementById('theme-icon');
            if (html.classList.contains('dark')) {
                html.classList.remove('dark');
                html.classList.add('light');
                icon.className = 'fa-solid fa-sun';
                showToast('Светлая тема');
            } else {
                html.classList.remove('light');
                html.classList.add('dark');
                icon.className = 'fa-solid fa-moon';
                showToast('Темная тема');
            }
        }

        function switchTab(tab) {
            ['dashboard', 'music', 'concierge', 'logs'].forEach(t => {
                document.getElementById(`tab-${t}`).classList.add('hidden');
                const btn = document.getElementById(`nav-${t}`);
                btn.classList.remove('dark:bg-white/10', 'bg-black/10');
                btn.classList.add('opacity-50');
            });
            document.getElementById(`tab-${tab}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`nav-${tab}`);
            activeBtn.classList.remove('opacity-50');
            activeBtn.classList.add('dark:bg-white/10', 'bg-black/10');
        }

        function toggleModeCheckbox(mode) {
            const checkbox = document.getElementById(`mode-${mode}`);
            checkbox.checked = !checkbox.checked;
            showToast(`Режим "${mode}" переключен`);
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').innerText = msg;
            toast.style.opacity = '1';
            setTimeout(() => {
                toast.style.opacity = '0';
            }, 2000);
        }
    </script>
